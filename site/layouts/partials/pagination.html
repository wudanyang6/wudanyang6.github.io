{{- /* 入参：dict "paginator" 。首页与 taxonomy 页共用。 窗口策略：始终显示首尾各 2 页 + 当前页 ±1，跳过的区间以 … 占位。 单页（TotalPages <= 1）时整块不渲染。 */ -}} {{- $p := .paginator -}} {{- if gt $p.TotalPages 1 }}
{{- with $p.Prev }} ← 上一页 {{- else }} ← 上一页 {{- end }} {{- $cur := $p.PageNumber -}} {{- $total := $p.TotalPages -}} {{- $prev := 0 -}} {{- range $p.Pagers -}} {{- $n := .PageNumber -}} {{- $show := or (le $n 2) (ge $n (sub $total 1)) (and (ge $n (sub $cur 1)) (le $n (add $cur 1))) -}} {{- if $show -}} {{- /* 与上一个已显示页码之间跳过的区间用 … 表示 */ -}} {{- if gt (sub $n $prev) 1 }}…{{ end -}} {{- if eq $n $cur -}} {{ $n }} {{- else -}} {{ $n }} {{- end -}} {{- $prev = $n -}} {{- end -}} {{- end }} {{- with $p.Next }} 下一页 → {{- else }} 下一页 → {{- end }} 第 {{ $p.PageNumber }} / {{ $p.TotalPages }} 页
{{- end }}
